Sudoku Hackoro Documentation

What is Sudoku Hackoro?

Sudoku Hackoro is an interactive web-based Sudoku game designed to challenge your problem-solving skills while providing AI assistance. The game features a modern interface, mobile-friendly layout, and advanced functionality to make playing Sudoku enjoyable and educational.

How to Play

  • Fill in the 9x9 grid so that each row, column, and 3x3 box contains the numbers 1 to 9.
  • Use the number-only inputs; invalid characters will automatically be removed.
  • Click the Validate button to check if your solution is correct.
  • Click 🤖 Solve with AI to watch the AI solve the puzzle step by step.
  • Click 🎲 New Puzzle to generate a new random Sudoku challenge.

Features

  • Dynamic 9x9 Sudoku grid with responsive layout for mobile and desktop.
  • AI solver with MRV (Minimum Remaining Value) and backtracking algorithm.
  • Step-by-step AI solving with color highlights for working and placed cells.
  • Timer showing seconds and milliseconds during AI solving.
  • Random puzzle generator with adjustable difficulty.
  • Player solution validation with alerts for correctness.
  • Log panel displaying AI decisions and backtracking steps.

Core AI Solver Code


// MRV (Minimum Remaining Value) to pick best empty cell
function findBestCell(board) {
    let best = null;
    let minOptions = 10;
    for (let r = 0; r < 9; r++) {
        for (let c = 0; c < 9; c++) {
            if (board[r][c] === 0) {
                let options = 0;
                for (let n = 1; n <= 9; n++)
                    if (isValid(board, r, c, n)) options++;
                if (options < minOptions) {
                    minOptions = options;
                    best = { r, c };
                }
            }
        }
    }
    return best;
}

// Recursive AI solver with backtracking
async function solve(board) {
    const best = findBestCell(board);
    if (!best) return true;
    const { r, c } = best;
    const cell = getCell(r, c);
    cell.classList.add("working");
    for (let num = 1; num <= 9; num++) {
        if (isValid(board, r, c, num)) {
            board[r][c] = num;
            cell.value = num;
            cell.classList.remove("working");
            cell.classList.add("placed");
            if (await solve(board)) return true;
            // backtrack
            board[r][c] = 0;
            cell.value = "";
            cell.classList.remove("placed");
            cell.classList.add("working");
        }
    }
    cell.classList.remove("working");
    return false;
}
